OpenCL Host API 基于 C 语言，所有对象通过引用计数管理。典型流程：发现平台/设备 → 创建 context 和命令队列 → 编译 kernel → 分配 buffer → 设置参数 → 执行 → 读回结果。

## 平台与设备发现

```c
#include <CL/cl.h>

cl_platform_id platform;
cl_device_id device;
cl_uint num_platforms, num_devices;

// 获取平台数量
clGetPlatformIDs(0, NULL, &num_platforms);

// 获取第一个平台
clGetPlatformIDs(1, &platform, NULL);

// 从平台上获取 GPU 设备
clGetDeviceIDs(platform, CL_DEVICE_TYPE_GPU, 1, &device, NULL);

// 查询平台/设备信息
char name[256];
clGetPlatformInfo(platform, CL_PLATFORM_NAME, sizeof(name), name, NULL);
clGetDeviceInfo(device, CL_DEVICE_NAME, sizeof(name), name, NULL);

// 查询设备关键参数
cl_uint max_cu, max_wg_size;
size_t max_wg[3], local_mem_size;
clGetDeviceInfo(device, CL_DEVICE_MAX_COMPUTE_UNITS, sizeof(max_cu), &max_cu, NULL);
clGetDeviceInfo(device, CL_DEVICE_MAX_WORK_GROUP_SIZE, sizeof(max_wg_size), &max_wg_size, NULL);
clGetDeviceInfo(device, CL_DEVICE_MAX_WORK_ITEM_SIZES, sizeof(max_wg), max_wg, NULL);
clGetDeviceInfo(device, CL_DEVICE_LOCAL_MEM_SIZE, sizeof(local_mem_size), &local_mem_size, NULL);
```

## Context 与 Command Queue

```c
cl_int err;
cl_context context;
cl_command_queue queue;

// 创建 context
context = clCreateContext(NULL, 1, &device, NULL, NULL, &err);

// 创建命令队列（OpenCL 2.0+ 支持 clCreateCommandQueueWithProperties）
queue = clCreateCommandQueue(context, device, 0, &err);

// 乱序执行 + profiling（可选）
cl_command_queue queue = clCreateCommandQueue(
    context, device, CL_QUEUE_OUT_OF_ORDER_EXEC_MODE_ENABLE | CL_QUEUE_PROFILING_ENABLE, &err);
```

## 编译 Kernel

```c
// Kernel 源码（字符串形式）
const char *source = R"(
__kernel void vec_add(__global const float *a,
                      __global const float *b,
                      __global float *c) {
    int gid = get_global_id(0);
    c[gid] = a[gid] + b[gid];
}
)";

// 从源码创建 program
cl_program program = clCreateProgramWithSource(context, 1, &source, NULL, &err);

// 编译
err = clBuildProgram(program, 1, &device, NULL, NULL, NULL);
if (err != CL_SUCCESS) {
    // 获取编译错误信息
    size_t log_size;
    clGetProgramBuildInfo(program, device, CL_PROGRAM_BUILD_LOG, 0, NULL, &log_size);
    char *log = (char *)malloc(log_size);
    clGetProgramBuildInfo(program, device, CL_PROGRAM_BUILD_LOG, log_size, log, NULL);
    printf("Build error:\n%s\n", log);
    free(log);
}

// 创建 kernel
cl_kernel kernel = clCreateKernel(program, "vec_add", &err);
```

编译选项：

```c
// 等价于：gcc -I. -DMACRO=1 -O2 -cl-mad-enable
clBuildProgram(program, 1, &device,
    "-I. -DMACRO=1 -cl-opt-disable -cl-mad-enable", NULL, NULL);
```

常用选项：`-cl-mad-enable`（允许 a*b+c 融合）、`-cl-fast-relaxed-math`（放宽数学精度）、`-Werror`。

## Kernel 参数设置与执行

```c
// 设置 kernel 参数
clSetKernelArg(kernel, 0, sizeof(cl_mem), &d_a);
clSetKernelArg(kernel, 1, sizeof(cl_mem), &d_b);
clSetKernelArg(kernel, 2, sizeof(cl_mem), &d_c);

// 添加一个标量参数
float scale = 2.0f;
clSetKernelArg(kernel, 3, sizeof(float), &scale);

// 添加 local memory 参数（运行时分配）
clSetKernelArg(kernel, 4, sizeof(float) * 256, NULL);

// 设置 NDRange 并执行
size_t global_size = N;
size_t local_size = 64;
clEnqueueNDRangeKernel(queue, kernel, 1, NULL, &global_size, &local_size, 0, NULL, NULL);
```

* `work_dim`：维度（1/2/3）
* `global_work_size`：全局 work-item 总数，必须能被 local_work_size 整除
* `local_work_size`：每个 work-group 的 work-item 数，为 `NULL` 时 OpenCL 自动选择
* `global_work_offset`：起始偏移（`NULL` 从 0 开始）

## 同步

```c
// 等待单个命令完成
clFinish(queue);

// 刷新所有排队命令（不等待完成）
clFlush(queue);

// 事件等待
cl_event write_event;
clEnqueueWriteBuffer(queue, d_a, CL_FALSE, 0, size, a, 0, NULL, &write_event);

// 后续命令等待 write_event 完成
clEnqueueNDRangeKernel(queue, kernel, 1, NULL, &global, &local, 1, &write_event, NULL);

// 等待事件
clWaitForEvents(1, &write_event);

// Profiling：获取命令耗时
cl_ulong start, end;
clGetEventProfilingInfo(write_event, CL_PROFILING_COMMAND_START, sizeof(start), &start, NULL);
clGetEventProfilingInfo(write_event, CL_PROFILING_COMMAND_END, sizeof(end), &end, NULL);
double elapsed_ms = (end - start) * 1e-6;
```

## 资源释放

```c
clReleaseKernel(kernel);
clReleaseProgram(program);
clReleaseMemObject(d_a);
clReleaseMemObject(d_b);
clReleaseMemObject(d_c);
clReleaseCommandQueue(queue);
clReleaseContext(context);
```

释放顺序：常用对象的依赖方向是 context → queue/mem/program → kernel，先释放依赖端再释放被依赖端。也可逆序释放——OpenCL 是引用计数，最终计数归零才释放。

## 错误处理

```c
#define CL_CHECK(err, msg) do { \
    if (err != CL_SUCCESS) { \
        fprintf(stderr, "Error %d: %s\n", err, msg); \
        exit(1); \
    } \
} while(0)

cl_int err;
cl_mem buf = clCreateBuffer(context, CL_MEM_READ_ONLY, size, NULL, &err);
CL_CHECK(err, "clCreateBuffer failed");
```

常见错误码：

| 错误码 | 含义 |
|--------|------|
| `CL_INVALID_VALUE` | 参数值不合法 |
| `CL_INVALID_BUFFER_SIZE` | buffer 大小超出设备限制 |
| `CL_INVALID_WORK_GROUP_SIZE` | local work size 不合法 |
| `CL_MEM_OBJECT_ALLOCATION_FAILURE` | 设备显存不足 |
| `CL_OUT_OF_RESOURCES` | 设备资源耗尽 |
| `CL_INVALID_KERNEL` | kernel 函数未找到 |
| `CL_BUILD_PROGRAM_FAILURE` | kernel 编译失败（检查 build log） |

## 完整示例：向量加法

```c
#include <CL/cl.h>
#include <stdio.h>
#include <stdlib.h>

#define N 1024
#define CHECK(err, msg) if (err != CL_SUCCESS) { printf("Error %d: %s\n", err, msg); exit(1); }

const char *source = R"(
__kernel void vec_add(__global const float *a,
                      __global const float *b,
                      __global float *c) {
    int gid = get_global_id(0);
    c[gid] = a[gid] + b[gid];
}
)";

int main() {
    cl_int err;
    float a[N], b[N], c[N];
    for (int i = 0; i < N; i++) { a[i] = i; b[i] = i * 2; }

    // 1. 平台与设备
    cl_platform_id platform;
    clGetPlatformIDs(1, &platform, NULL);
    cl_device_id device;
    clGetDeviceIDs(platform, CL_DEVICE_TYPE_GPU, 1, &device, NULL);

    // 2. Context 和命令队列
    cl_context context = clCreateContext(NULL, 1, &device, NULL, NULL, &err);
    CHECK(err, "clCreateContext");
    cl_command_queue queue = clCreateCommandQueue(context, device, 0, &err);
    CHECK(err, "clCreateCommandQueue");

    // 3. 编译 kernel
    cl_program program = clCreateProgramWithSource(context, 1, &source, NULL, &err);
    CHECK(err, "clCreateProgramWithSource");
    err = clBuildProgram(program, 1, &device, NULL, NULL, NULL);
    CHECK(err, "clBuildProgram");
    cl_kernel kernel = clCreateKernel(program, "vec_add", &err);
    CHECK(err, "clCreateKernel");

    // 4. 创建 buffer
    cl_mem d_a = clCreateBuffer(context, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR,
                                sizeof(float)*N, a, &err);
    cl_mem d_b = clCreateBuffer(context, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR,
                                sizeof(float)*N, b, &err);
    cl_mem d_c = clCreateBuffer(context, CL_MEM_WRITE_ONLY, sizeof(float)*N, NULL, &err);

    // 5. 设置参数
    clSetKernelArg(kernel, 0, sizeof(cl_mem), &d_a);
    clSetKernelArg(kernel, 1, sizeof(cl_mem), &d_b);
    clSetKernelArg(kernel, 2, sizeof(cl_mem), &d_c);

    // 6. 执行
    size_t global = N, local = 64;
    clEnqueueNDRangeKernel(queue, kernel, 1, NULL, &global, &local, 0, NULL, NULL);

    // 7. 读回结果
    clEnqueueReadBuffer(queue, d_c, CL_TRUE, 0, sizeof(float)*N, c, 0, NULL, NULL);

    // 8. 验证
    int correct = 1;
    for (int i = 0; i < N; i++)
        if (c[i] != a[i] + b[i]) { correct = 0; break; }
    printf(correct ? "Passed\n" : "Failed\n");

    // 9. 清理
    clReleaseMemObject(d_a); clReleaseMemObject(d_b); clReleaseMemObject(d_c);
    clReleaseKernel(kernel); clReleaseProgram(program);
    clReleaseCommandQueue(queue); clReleaseContext(context);
    return 0;
}
```

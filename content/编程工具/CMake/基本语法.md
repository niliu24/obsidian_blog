> CMake 脚本语言是一种**声明式 + 命令式**混合的 DSL，以 `CMakeLists.txt` 为构建定义文件。命令不区分大小写，但通常使用小写。变量名大小写敏感。

## CMakeLists.txt 基本结构

每个 CMake 项目从顶层 `CMakeLists.txt` 开始。典型结构：

```cmake
# 最小版本要求（必须放在 project() 之前）
cmake_minimum_required(VERSION 3.16)

# 项目名称和语言
project(MyProject VERSION 1.0.0 LANGUAGES CXX C)

# 全局配置
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 添加子目录
add_subdirectory(lib)
add_subdirectory(app)

# 测试
enable_testing()
add_test(NAME my_test COMMAND my_app)
```

### `cmake_minimum_required`

```cmake
cmake_minimum_required(VERSION <版本号> [FATAL_ERROR])
```

* **必须**放在 `project()` 之前
* 指定 CMake 的最低版本，版本低于此则报错退出
* 同时设定 `CMAKE_POLICY_DEFAULT_${POLICY}` 的行为为 `NEW`

### `project()`

```cmake
project(<项目名>
  [VERSION <主>.<次>.<补丁>.<微>]
  [DESCRIPTION <描述>]
  [HOMEPAGE_URL <网址>]
  [LANGUAGES <语言列表>])
```

设置项目名称并自动定义变量：

| 变量 | 值 |
|------|-----|
| `PROJECT_NAME` | 项目名 |
| `PROJECT_SOURCE_DIR` | 包含当前 `project()` 的 `CMakeLists.txt` 所在目录 |
| `PROJECT_BINARY_DIR` | 对应的构建目录 |
| `CMAKE_PROJECT_NAME` | 顶层 `project()` 定义的项目名 |
| `<项目名>_VERSION` | 版本号各组件（`_MAJOR`、`_MINOR`、`_PATCH`、`_TWEAK`） |

## 注释

```cmake
# 单行注释，以 # 开头

# CMake 没有块注释语法，但可以用 if(FALSE) 模拟
if(FALSE)
    这里的内容不会被 CMake 执行
    但不能包含不配对的引号
endif()
```

## 变量

### 普通变量

```cmake
# 赋值
set(MY_VAR "hello")
set(MY_VAR hello)                # 可以省略引号（不含空格时）

# 引用
message(STATUS "MY_VAR = ${MY_VAR}")

# 列表赋值
set(LIST_VAR a b c)              # 分号分隔或空格分隔
set(LIST_VAR "a;b;c")            # 等价表示

# 追加到列表
list(APPEND LIST_VAR "d" "e")

# 获取列表长度
list(LENGTH LIST_VAR len)
message(STATUS "len = ${len}")

# 获取列表元素
list(GET LIST_VAR 0 first)
message(STATUS "first = ${first}")
```

### 环境变量

```cmake
# 读取环境变量
message(STATUS "PATH = $ENV{PATH}")

# 设置环境变量（仅影响当前 CMake 进程及其子进程）
set(ENV{MY_ENV} "some_value")
```

### Cache 变量

Cache 变量存储在 `CMakeCache.txt` 中，跨 CMake 运行保持不变。用于用户可配置的选项。

```cmake
# option — 典型的 Cache 布尔变量
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(USE_OPENMP "Enable OpenMP support" ON)

# set CACHE — 显式定义 Cache 变量
set(MY_CACHE_STRING "default" CACHE STRING "A cache variable")
set(MY_CACHE_PATH "/usr/local" CACHE PATH "Install prefix")
set(MY_CACHE_FILE "/etc/config.ini" CACHE FILEPATH "Config file path")
set(MY_CACHE_BOOL ON CACHE BOOL "A boolean option")

# Cache 变量可以加 FORCE 强制覆盖已有值
set(MY_CACHE_STRING "forced" CACHE STRING "Forced value" FORCE)

# 标记为高级（不在 cmake-gui 默认显示）
mark_as_advanced(MY_CACHE_STRING)
```

| Cache 变量类型 | 说明 | cmake-gui 控件 |
|---------------|------|----------------|
| `BOOL` | ON/OFF | 复选框 |
| `STRING` | 任意字符串 | 文本输入 |
| `PATH` | 目录路径 | 目录选择器 |
| `FILEPATH` | 文件路径 | 文件选择器 |
| `INTERNAL` | 不公开，用于缓存计算值 | 不显示 |

### 变量作用域

CMake 变量作用域按**目录**划分。子目录通过 `add_subdirectory()` 或 `include()` 引入，拥有自己的作用域：

```cmake
# 顶层 CMakeLists.txt
set(TOP_VAR "top")
add_subdirectory(sub)

# sub/CMakeLists.txt
message(STATUS "${TOP_VAR}")   # 可以读取父目录变量
set(SUB_VAR "sub")
# 返回父目录后 SUB_VAR 不可见
```

| 方式 | 子目录能否读取父变量 | 子目录修改变量是否影响父目录 |
|------|---------------------|---------------------------|
| `add_subdirectory()` | 能（继承） | 否（除非 `PARENT_SCOPE`） |
| `include()` | 能（当前作用域展开） | 是（在当前作用域执行） |
| `function()` | 能 | 否（除非 `PARENT_SCOPE`） |
| `macro()` | 能（原地文本替换） | 是 |

## 控制流

### 条件判断：`if/elseif/else/endif`

```cmake
if(<条件>)
    # ...
elseif(<条件>)
    # ...
else()
    # ...
endif()
```

条件判断规则：

```cmake
# 常量真值
if(ON)        → TRUE
if(YES)       → TRUE
if(TRUE)      → TRUE
if(1)         → TRUE
if(Y)         → TRUE
if(NONZERO)   → TRUE  # 非零数字字符串

# 常量假值
if(OFF)       → FALSE
if(NO)        → FALSE
if(FALSE)     → FALSE
if(0)         → FALSE
if(N)         → FALSE
if(IGNORE)    → FALSE
if(NOTFOUND)  → FALSE
if("")        → FALSE  # 空字符串

# 变量：当变量定义为非假值时求值为真
set(HAVE_FOO ON)
if(HAVE_FOO)          # → TRUE

set(BAR OFF)
if(BAR)               # → FALSE

# 定义检查
if(DEFINED MY_VAR)
    message("MY_VAR is defined")
endif()

# 比较操作符
if(MY_VAR STREQUAL "hello")          # 字符串相等
if(MY_VAR MATCHES "^hello.*")        # 正则匹配
if(VERSION VERSION_GREATER 3.0)      # 版本比较
if(COUNT GREATER 10)                 # 数值比较
if(COUNT LESS_EQUAL 5)               # 数值比较

# 复合条件
if(USE_OPENMP AND OPENMP_FOUND)
if(USE_CUDA OR USE_HIP)
if(NOT MSVC)
if(UNIX AND NOT APPLE)               # Linux 检测
```

### 循环：`foreach/endforeach`

```cmake
# 遍历列表
set(SOURCES main.cpp util.cpp render.cpp)
foreach(src ${SOURCES})
    message(STATUS "Source: ${src}")
endforeach()

# 遍历范围
foreach(i RANGE 1 10)
    message(STATUS "i = ${i}")
endforeach()

# 遍历带步长
foreach(i RANGE 0 10 2)
    message(STATUS "even: ${i}")
endforeach()

# 遍历 zip（CMake 3.17+）
foreach(src header IN ZIP_LISTS SOURCES HEADERS)
    message(STATUS "${src_0} → ${src_1}")
endforeach()
```

### 循环：`while/endwhile`

```cmake
set(counter 5)
while(counter GREATER 0)
    message(STATUS "countdown: ${counter}")
    math(EXPR counter "${counter} - 1")
endwhile()
```

## 函数与宏

### function / endfunction

函数创建**局部作用域**——任何 `set()` 都不影响调用者作用域，除非使用 `PARENT_SCOPE`。

```cmake
function(add_and_return OUT_VAR A B)
    math(EXPR result "${A} + ${B}")

    # ARGV / ARGN — 所有参数 / 超出命名参数的部分
    message(STATUS "Named args: ${A}, ${B}")
    message(STATUS "Total args: ${ARGC}")
    message(STATUS "Extra args: ${ARGN}")

    # 通过 PARENT_SCOPE "返回" 值
    set(${OUT_VAR} ${result} PARENT_SCOPE)
endfunction()

add_and_return(sum 3 5)
message(STATUS "sum = ${sum}")  # → 8

# 关键字参数模式（CMake 3.18+ 推荐 cmake_parse_arguments）
function(my_target TARGET_NAME)
    set(options STATIC SHARED)
    set(oneValueArgs VERSION)
    set(multiValueArgs SOURCES DEPENDS)
    cmake_parse_arguments(MY "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

    if(MY_STATIC)
        add_library(${TARGET_NAME} STATIC ${MY_SOURCES})
    elseif(MY_SHARED)
        add_library(${TARGET_NAME} SHARED ${MY_SOURCES})
    endif()

    if(MY_VERSION)
        set_target_properties(${TARGET_NAME} PROPERTIES VERSION ${MY_VERSION})
    endif()
    if(MY_DEPENDS)
        target_link_libraries(${TARGET_NAME} PUBLIC ${MY_DEPENDS})
    endif()
endfunction()

my_target(mylib
    STATIC
    VERSION 1.0
    SOURCES a.cpp b.cpp
    DEPENDS fmt)
```

### macro / endmacro

宏是**文本替换**——参数在调用点展开，不创建新作用域。宏内 `set()` 直接影响调用者作用域。

```cmake
macro(print_message MSG)
    message(STATUS "[${PROJECT_NAME}] ${MSG}")
    set(last_msg "${MSG}")    # 影响调用者作用域！
endmacro()

print_message("Building targets...")
message(STATUS "Last: ${last_msg}")  # 可以访问到
```

### 函数 vs 宏

| 维度 | function | macro |
|------|----------|-------|
| **作用域** | 局部（`PARENT_SCOPE` 可返回） | 在调用者作用域内展开 |
| **参数** | 按值传递副本 | 文本替换（类似 C 宏） |
| `set()` 影响 | 函数内局部 | 修改调用者变量 |
| **return()** | 支持提前返回 | 不支持 |
| **推荐场景** | 绝大多数情况 | 需要修改调用者变量或性能敏感微调 |

## 消息打印

```cmake
message(STATUS "信息性消息（默认前缀 --）")
message("普通消息（无前缀，直接输出）")
message(WARNING "警告（黄色，继续执行）")
message(AUTHOR_WARNING "开发者警告（仅 -Wdev 时显示）")
message(SEND_ERROR "错误（继续配置，但跳过生成")
message(FATAL_ERROR "致命错误（立即停止）")
message(DEPRECATION "弃用警告（可由策略控制）")
```

## Generator Expressions

Generator Expressions 在**构建时**（而非配置时）求值，语法为 `$<…>`。它们常用于目标属性的条件设置，因为配置时还不知道编译器和平台细节。

```cmake
# 条件表达式
target_compile_definitions(myapp PRIVATE
    $<$<CONFIG:Debug>:DEBUG_BUILD>         # Debug 配置定义 DEBUG_BUILD
    $<$<CONFIG:Release>:NDEBUG>            # Release 配置定义 NDEBUG
)

# 平台判断
target_sources(myapp PRIVATE
    $<$<PLATFORM_ID:Linux>:linux_specific.cpp>
    $<$<PLATFORM_ID:Windows>:win_specific.cpp>
)

# 编译语言判断
target_compile_options(myapp PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-std=c++17>
    $<$<COMPILE_LANGUAGE:C>:-std=c11>
)

# 目标存在判断
target_link_libraries(myapp PRIVATE
    $<$<TARGET_EXISTS:fmt>:fmt>            # 仅当 fmt 目标存在时链接
)

# 字符串操作
set(version "1.2.3")
message(STATUS "$<VERSION_GREATER:${version},1.0.0>")  # → 1 (true)

# 布尔表达式嵌套
$<$<AND:$<CONFIG:Debug>,$<PLATFORM_ID:Linux>>:linux_debug_flag>

# 列表展开
$<JOIN:$<TARGET_PROPERTY:myapp,SOURCES>,;>  # 用分号连接源文件列表

# 编译特性
target_compile_features(myapp PRIVATE cxx_std_17)
```

### 常用 Generator Expressions

| 表达式 | 说明 |
|--------|------|
| `$<CONFIG:cfg>` | 配置名为 cfg 时求值 1 |
| `$<PLATFORM_ID:id>` | 平台 ID 匹配时求值 1 |
| `$<COMPILE_LANGUAGE:lang>` | 编译语言匹配时求值 1 |
| `$<TARGET_EXISTS:tgt>` | 目标存在时求值 1 |
| `$<TARGET_PROPERTY:tgt,prop>` | 获取目标属性值 |
| `$<BOOL:string>` | 字符串转换为布尔 0/1 |
| `$<STREQUAL:a,b>` | 字符串相等检查 |
| `$<IN_LIST:a,list>` | 元素是否在列表中 |
| `$<JOIN:list,sep>` | 用分隔符连接列表 |
| `$<BUILD_INTERFACE:…>` | 构建目录内有效 |
| `$<INSTALL_INTERFACE:…>` | 安装目录内有效 |
| `$<LOWER_CASE:str>` | 转小写 |
| `$<UPPER_CASE:str>` | 转大写 |
| `$<GENEX_EVAL:expr>` | 递归求值 Generator Expression |

## 完整示例

一个展示核心语法的完整 `CMakeLists.txt`：

```cmake
cmake_minimum_required(VERSION 3.20)
project(SyntaxDemo VERSION 1.0.0 LANGUAGES CXX)

# === 变量 ===
set(APP_NAME "SyntaxDemoApp")
set(SOURCES main.cpp helper.cpp)

option(ENABLE_LOGGING "Enable verbose logging" OFF)
set(BUILD_TESTS ON CACHE BOOL "Build unit tests")

# === 条件判断 ===
if(ENABLE_LOGGING)
    add_compile_definitions(LOGGING_ENABLED)
    message(STATUS "Logging enabled")
else()
    message(STATUS "Logging disabled")
endif()

# === 函数定义 ===
function(add_executable_with_options TARGET_NAME)
    set(options CONSOLE GUI)
    set(oneValueArgs VERSION)
    set(multiValueArgs SOURCES LIBS)
    cmake_parse_arguments(ARG "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

    add_executable(${TARGET_NAME} ${ARG_SOURCES})

    if(ARG_VERSION)
        set_target_properties(${TARGET_NAME} PROPERTIES
            VERSION ${ARG_VERSION}
            SOVERSION ${ARG_VERSION_MAJOR})
    endif()

    if(ARG_LIBS)
        target_link_libraries(${TARGET_NAME} PRIVATE ${ARG_LIBS})
    endif()

    # Generator Expression: 不同配置加上不同编译选项
    target_compile_definitions(${TARGET_NAME} PRIVATE
        $<$<CONFIG:Debug>:DEBUG_MODE>
        $<$<CONFIG:Release>:RELEASE_MODE>)
endfunction()

# === 使用函数 ===
add_executable_with_options(${APP_NAME}
    VERSION 1.0.0
    SOURCES ${SOURCES}
    LIBS fmt spdlog)

# === 循环 ===
set(SUBDIRS src lib tests)
foreach(dir ${SUBDIRS})
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}/CMakeLists.txt")
        message(STATUS "Adding subdirectory: ${dir}")
        add_subdirectory(${dir})
    endif()
endforeach()

# === 测试 ===
if(BUILD_TESTS)
    enable_testing()
    foreach(i RANGE 1 5)
        add_test(NAME test_${i} COMMAND ${APP_NAME} --test ${i})
    endforeach()
endif()
```

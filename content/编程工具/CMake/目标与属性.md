> CMake 中 **Target**（目标）是构建产物的抽象表示——可执行文件、库、自定义命令等。Modern CMake（3.0+）以目标为核心，通过 `PUBLIC/PRIVATE/INTERFACE` 关键字精确控制属性的传递性，取代了旧式的全局命令。

## Target 类型

### `add_executable` — 可执行文件

```cmake
# 普通可执行文件
add_executable(myapp main.cpp util.cpp)

# 导入的可执行文件（不构建，仅引用）
add_executable(some_tool IMPORTED)
set_target_properties(some_tool PROPERTIES
    IMPORTED_LOCATION /usr/bin/some_tool)

# 别名（cmake 3.11+）
add_executable(myapp::myapp ALIAS myapp)
```

常用属性：

```cmake
set_target_properties(myapp PROPERTIES
    OUTPUT_NAME "my_app"           # 输出文件名
    WIN32_EXECUTABLE TRUE          # Windows GUI 应用（无控制台窗口）
    MACOSX_BUNDLE TRUE             # macOS App Bundle
    CXX_STANDARD 17                # C++ 标准
    CXX_STANDARD_REQUIRED ON)
```

### `add_library` — 库

```cmake
# 静态库
add_library(mylib STATIC a.cpp b.cpp)

# 动态库/共享库
add_library(mylib SHARED a.cpp b.cpp)

# MODULE 库 — 运行时动态加载（不参与链接），如插件
add_library(myplugin MODULE plugin.cpp)

# INTERFACE 库 — 仅头文件库或纯接口，没有编译产物
add_library(myheader_only INTERFACE)
target_include_directories(myheader_only INTERFACE include/)

# OBJECT 库 — 编译为目标文件但不归档/链接，可用于多个目标
add_library(myobjects OBJECT obj1.cpp obj2.cpp)
add_executable(main1 main1.cpp $<TARGET_OBJECTS:myobjects>)
add_executable(main2 main2.cpp $<TARGET_OBJECTS:myobjects>)
```

| 库类型 | 产物 | 链接方式 | 典型用途 |
|--------|------|----------|----------|
| `STATIC` | `.a` / `.lib` | 链接时复制到可执行文件 | 内部模块、减小分发体积 |
| `SHARED` | `.so` / `.dylib` / `.dll` | 运行时加载 | 公共库、插件 |
| `MODULE` | `.so` / `.dll` | `dlopen` / `LoadLibrary` 动态加载 | 插件系统 |
| `INTERFACE` | 无 | 仅传播编译要求 | 仅头文件库 |
| `OBJECT` | `.o` / `.obj` | 直接链接到其他目标 | 避免重复编译 |

### `add_custom_target` — 自定义目标

```cmake
# 始终执行的伪目标
add_custom_target(run
    COMMAND ${CMAKE_CURRENT_BINARY_DIR}/myapp
    DEPENDS myapp
    COMMENT "Running myapp..."
)

# 依赖其他目标的伪目标
add_custom_target(docs
    COMMAND doxygen Doxyfile
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Generating documentation..."
    DEPENDS
        ${CMAKE_CURRENT_SOURCE_DIR}/Doxyfile
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# 添加依赖，使所有目标都依赖此自定目标
add_dependencies(myapp run)
```

## 目标属性与传递性

Modern CMake 的核心哲学：属性通过 `PUBLIC/PRIVATE/INTERFACE` 在目标间**自动传播**。

### `target_include_directories`

```cmake
target_include_directories(mylib
    PUBLIC                           # 自己和链接者都能用
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
    PRIVATE                          # 仅自己可用
        ${CMAKE_CURRENT_SOURCE_DIR}/src
        ${CMAKE_CURRENT_SOURCE_DIR}/internal
)
```

| 关键字 | 影响自己 | 影响链接者 | 影响链接者的链接者 |
|--------|----------|-----------|-------------------|
| `PRIVATE` | 是 | 否 | 否 |
| `PUBLIC` | 是 | 是 | 传递 |
| `INTERFACE` | 否 | 是 | 传递 |

### `target_compile_definitions`

```cmake
target_compile_definitions(mylib
    PRIVATE
        MYLIB_BUILDING    # 编译库本身时定义
    PUBLIC
        MYLIB_EXPORTS     # 使用库时也需要
    INTERFACE
        USING_MYLIB       # 仅使用者需要
)
```

### `target_compile_options`

```cmake
target_compile_options(mylib
    PRIVATE
        -Wall -Wextra -Werror
    INTERFACE
        $<$<CXX_COMPILER_ID:MSVC>:/W4>
        $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra>
)
```

### `target_link_libraries` — 传递依赖

这是 Modern CMake 最强大的特性——依赖自动传播。

```cmake
# mylib 直接依赖
target_link_libraries(mylib
    PUBLIC
        fmt                          # 使用者的公开 API 依赖 fmt
    PRIVATE
        spdlog                       # 仅内部使用
        zlib::zlib                   # 实现细节
    INTERFACE
        Eigen3::Eigen                # 仅头文件依赖
)

# myapp 使用 mylib，自动继承所有 PUBLIC/INTERFACE 传递属性
target_link_libraries(myapp PRIVATE mylib)
# 等价于手动：
#   target_include_directories(myapp ...)  # mylib 的 PUBLIC include
#   target_link_libraries(myapp ...)       # fmt
#   target_compile_definitions(myapp ...)  # MYLIB_EXPORTS（如有）
```

### 传递性规则图解

```
mylib (PUBLIC: fmt) ──→ myapp
    │                       │
    ▼                       ▼
mylib 编译时用 fmt      myapp 编译时也自动用 fmt
                         （PUBLIC 传播）

mylib (PRIVATE: spdlog) ──→ myapp
    │
    ▼
mylib 编译时用 spdlog      myapp 完全看不到 spdlog
```

### `target_sources`

```cmake
# 现代 CMake 推荐在 CMakeLists.txt 中加入源文件
add_library(mylib STATIC)

# 按模块分组添加源文件
target_sources(mylib PRIVATE
    src/core.cpp
    src/utils.cpp
    src/render.cpp
)

# 条件编译
target_sources(mylib PRIVATE
    $<$<PLATFORM_ID:Linux>:src/linux_specific.cpp>
    $<$<PLATFORM_ID:Windows>:src/win_specific.cpp>
)
```

### `target_compile_features`

```cmake
# 要求编译器支持 C++17
target_compile_features(mylib PUBLIC cxx_std_17)

# 需要的具体特性
target_compile_features(mylib PRIVATE
    cxx_constexpr             # C++11 constexpr
    cxx_auto_type             # auto 关键字
    cxx_lambdas               # Lambda 表达式
)

# 链接者继承编译特性要求
target_link_libraries(myapp PRIVATE mylib)
# myapp 自动继承 cxx_std_17
```

## `set_target_properties`

```cmake
set_target_properties(mylib PROPERTIES
    # 输出名称
    OUTPUT_NAME "my_library"

    # 版本管理
    VERSION 2.1.0
    SOVERSION 2

    # 编译位置无关代码（默认 SHARED 和 MODULE 自动启用）
    POSITION_INDEPENDENT_CODE ON

    # 可见性
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON

    # 调试符号
    ENABLE_EXPORTS ON

    # 链接器
    LINKER_LANGUAGE CXX

    # 包含目录（旧式，不推荐新项目使用）
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include"
)
```

## `install(TARGETS)`

```cmake
# 安装目标到指定路径
install(TARGETS mylib myapp
    EXPORT MyProjectTargets           # 导出目标（供 find_package 使用）
    LIBRARY DESTINATION lib           # 共享库 → ${prefix}/lib
    ARCHIVE DESTINATION lib           # 静态库/导入库 → ${prefix}/lib
    RUNTIME DESTINATION bin           # 可执行文件 → ${prefix}/bin
    INCLUDES DESTINATION include      # 编译器接口包含路径
)

# 安装头文件
install(DIRECTORY include/
    DESTINATION include
    FILES_MATCHING PATTERN "*.h"
)

# 生成 CMake 包配置文件
install(EXPORT MyProjectTargets
    FILE MyProjectTargets.cmake
    DESTINATION lib/cmake/MyProject
)

# 生成版本配置文件
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/MyProjectConfigVersion.cmake"
    VERSION 1.0.0
    COMPATIBILITY AnyNewerVersion
)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/MyProjectConfigVersion.cmake"
    DESTINATION lib/cmake/MyProject
)
```

### `$<BUILD_INTERFACE>` 和 `$<INSTALL_INTERFACE>`

这两个 Generator Expressions 用于区分**构建时**和**安装后**的路径差异：

```cmake
target_include_directories(mylib
    PUBLIC
        # 构建时：使用源码目录下的 include
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        # 安装后：使用安装前缀下的 include
        $<INSTALL_INTERFACE:include>
)
```

* **构建时**：`$<BUILD_INTERFACE:xxx>` → `xxx`，`$<INSTALL_INTERFACE:yyy>` → 空
* **安装时**：`$<BUILD_INTERFACE:xxx>` → 空，`$<INSTALL_INTERFACE:yyy>` → `yyy`

## 完整示例：多目标项目

```cmake
cmake_minimum_required(VERSION 3.20)
project(GeometryLib VERSION 1.0.0 LANGUAGES CXX)

# === 编译选项 ===
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(BUILD_EXAMPLES "Build examples" ON)
option(BUILD_TESTING "Build tests" ON)

# === 库目标 ===
add_library(geometry
    src/shape.cpp
    src/transform.cpp
    src/matrix.cpp
)

set_target_properties(geometry PROPERTIES
    VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION_MAJOR}
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
)

target_include_directories(geometry
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
)

target_compile_features(geometry PUBLIC cxx_std_17)

# 传递依赖
target_link_libraries(geometry
    PUBLIC
        Eigen3::Eigen         # 公开 API 使用 Eigen
    PRIVATE
        fmt::fmt              # 内部实现使用 fmt
)

# === 可执行文件 ===
if(BUILD_EXAMPLES)
    add_executable(geometry_demo
        examples/demo.cpp
        examples/benchmark.cpp
    )

    target_link_libraries(geometry_demo PRIVATE geometry)

    # 不同配置不同定义
    target_compile_definitions(geometry_demo PRIVATE
        $<$<CONFIG:Debug>:DEMO_DEBUG>
    )

    install(TARGETS geometry_demo
        RUNTIME DESTINATION bin
    )
endif()

# === 安装规则 ===
install(TARGETS geometry
    EXPORT GeometryTargets
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
    INCLUDES DESTINATION include
)

install(DIRECTORY include/
    DESTINATION include
    FILES_MATCHING PATTERN "*.h"
)

install(EXPORT GeometryTargets
    FILE GeometryTargets.cmake
    DESTINATION lib/cmake/Geometry
)

# === 生成包配置文件 ===
include(CMakePackageConfigHelpers)
configure_package_config_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GeometryConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/GeometryConfig.cmake"
    INSTALL_DESTINATION lib/cmake/Geometry
)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/GeometryConfigVersion.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/GeometryConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/GeometryConfigVersion.cmake"
    DESTINATION lib/cmake/Geometry
)

# === 第三方目标示例 ===
# 导入已安装的库作为目标
add_library(ext_tool STATIC IMPORTED)
set_target_properties(ext_tool PROPERTIES
    IMPORTED_LOCATION /opt/ext/lib/libext_tool.a
    INTERFACE_INCLUDE_DIRECTORIES /opt/ext/include
)

# 仅头文件的目标
add_library(my_macros INTERFACE)
target_compile_definitions(my_macros INTERFACE
    USE_OPTIMIZED_MATH
)
target_link_libraries(geometry PRIVATE my_macros)
```

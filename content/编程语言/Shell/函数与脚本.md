## 函数与脚本

### 函数定义

Bash 函数有两种等价的定义方式：

```bash
# 方式一：function 关键字
function my_func {
    echo "Hello"
}

# 方式二：POSIX 兼容（推荐）
my_func() {
    echo "Hello"
}

# 单行定义
my_func() { echo "Hello"; }
```

### 函数参数

函数内部使用 `$1` `$2` 等特殊变量接收参数，（但不影响脚本的 `$0` `$1` 等参数）：

```bash
greet() {
    local name="$1"
    local greeting="${2:-Hello}"    # 默认值
    
    echo "$greeting, $name!"
    echo "函数收到 $# 个参数"
}

greet "Alice"             # Hello, Alice!
greet "Bob" "Hi"          # Hi, Bob!
greet "Charlie" "Hey" "extra"  # 多余参数被忽略
```

**参数处理最佳实践**：

```bash
process_args() {
    # 检查必要参数
    if (( $# < 2 )); then
        echo "用法: $FUNCNAME <input> <output>" >&2
        return 1
    fi
    
    local input="$1"
    local output="$2"
    local verbose="${3:-false}"
    
    # 使用 shift 处理不定参数
}

# 使用 getopts 处理选项
parse_opts() {
    while getopts ":f:o:v" opt; do
        case $opt in
            f) input="$OPTARG" ;;
            o) output="$OPTARG" ;;
            v) verbose=true ;;
            \?) echo "无效选项: -$OPTARG" >&2; return 1 ;;
        esac
    done
    shift $((OPTIND - 1))
}
```

### 返回值

函数有两种 " 返回 " 方式：

| 方式 | 机制 | 可返回范围 | 用法 |
|------|------|-----------|------|
| `return` | 设置退出码 | 0-255 | `return 0`（成功），`return 1`（失败） |
| `echo` | 输出到 stdout | 任意字符串 | `result=$(my_func)` 捕获输出 |

```bash
# return：返回退出码
is_even() {
    (( $1 % 2 == 0 )) && return 0 || return 1
}

if is_even 4; then
    echo "是偶数"
fi

# echo：返回数据
get_config() {
    local key="$1"
    local file="${2:-/etc/myapp.conf}"
    
    grep "^${key}=" "$file" | cut -d= -f2
}

db_host=$(get_config "db_host")
db_port=$(get_config "db_port")
echo "连接 ${db_host}:${db_port}"

# 返回数组
get_users() {
    local users=()
    while IFS=: read -r user _; do
        users+=("$user")
    done < /etc/passwd
    echo "${users[@]}"
}

users=($(get_users))    # 注意：包含空格的用户名会有问题
```

### 局部变量

`local` 使变量作用域限于函数内，避免污染全局命名空间：

```bash
global_var="全局"

my_func() {
    local local_var="局部"
    global_var="被函数修改"
    echo "内部: $local_var"     # 局部
}

my_func
echo "$global_var"              # 被函数修改
echo "$local_var"               # 空（函数外不可见）
```

**声明局部变量的最佳实践**：函数内所有变量都应使用 `local` 声明，包括循环变量：

```bash
process_items() {
    local items=("$@")
    local item result
    
    for item in "${items[@]}"; do
        result=$(process "$item")
        echo "$result"
    done
}
```

### source 与点命令

`source`（或等价的 `.`）在当前 Shell 环境中执行脚本，子脚本可以修改当前 Shell 的环境：

```bash
# 加载配置文件
source /etc/myapp.conf
. ~/.bashrc            # 点命令的等效写法

# 配置文件示例 (config.sh)
DB_HOST="localhost"
DB_PORT=3306
DB_USER="app"

# 主脚本
source config.sh
mysql -h "$DB_HOST" -P "$DB_PORT" -u "$DB_USER"

# source 和子脚本的区别
# test.sh 内容: export FOO=bar
./test.sh              # 子进程运行，FOO 不会传递到当前 Shell
source test.sh         # 当前进程运行，FOO 在当前 Shell 中可用
```

### 脚本调试

Bash 提供多种调试手段：

```bash
# 整脚本调试（放在 shebang 后或文件开头）
#!/bin/bash -x          # 执行时打印每条命令
#!/bin/bash -e          # 任何命令失败立即退出
#!/bin/bash -u          # 使用未定义变量时报错
#!/bin/bash -o pipefail # 管道中任何命令失败都算失败

# 运行时开启/关闭
set -x                  # 开启追踪模式（打印每条命令 + 展开后的参数）
set +x                  # 关闭追踪模式

set -e                  # errexit：任何非 0 退出码立即退出
set -u                  # nounset：使用未定义变量时报错
set -o pipefail         # 管道中最后一个命令失败不算成功

# 组合使用（推荐用于生产脚本）
set -euo pipefail
```

**调试示例**：

```bash
#!/bin/bash
set -euo pipefail       # 严格模式

debug() {
    [ "${DEBUG:-false}" = "true" ]( "${DEBUG:-false}" = "true" .md) && set -x
}

debug

# 逐行追踪
for i in {1..3}; do
    echo "debug: i=$i"
done

set +x                  # 关闭追踪

# 检查语法（不执行）
# bash -n script.sh

# 详细输出（执行但不显示展开过程）
# bash -v script.sh
```

### 信号处理（trap）

`trap` 捕获信号并在信号发生时执行指定命令：

```bash
# 清理操作
cleanup() {
    echo "执行清理..."
    rm -f /tmp/temp_$$
    exit 0
}

trap cleanup EXIT       # 脚本退出时执行（正常退出也会触发）
trap cleanup INT        # Ctrl+C 时执行
trap cleanup TERM       # kill 默认信号

# 忽略信号
trap '' INT             # 忽略 Ctrl+C

# 重置信号处理
trap - INT              # 恢复默认行为

# 常用信号
trap 'echo "收到 SIGINT, 退出"; exit 1' INT
trap 'echo "收到 SIGTERM, 优雅关闭"; graceful_shutdown; exit 0' TERM
trap 'echo "收到 SIGHUP, 重新加载配置"; reload_config' HUP
```

**常见信号表**：

| 信号 | 编号 | 默认行为 | 触发方式 |
|------|------|---------|---------|
| EXIT | 0 | - | 脚本退出（trap 特有） |
| INT | 2 | 终止进程 | Ctrl+C |
| QUIT | 3 | 终止 + 生成 core | Ctrl+\ |
| TERM | 15 | 终止进程 | `kill PID` 默认信号 |
| HUP | 1 | 终止进程 | 挂断/重载配置 |
| KILL | 9 | 强制终止 | 不可捕获/忽略 |
| STOP | 19 | 暂停进程 | 不可捕获/忽略 |

### 错误处理模式

```bash
#!/bin/bash
set -euo pipefail

# 错误处理函数
error_exit() {
    local msg="$1"
    local line="$2"
    echo "[错误] 第 $line 行: $msg" >&2
    exit 1
}

# 封装命令执行
run_cmd() {
    local cmd="$*"
    echo "[执行] $cmd"
    if ! eval "$cmd"; then
        error_exit "命令失败: $cmd" "$LINENO"
    fi
}

run_cmd "mkdir -p /tmp/build"
run_cmd "cp src/*.c /tmp/build/"
```

**模式：重试与超时**：

```bash
retry() {
    local max_attempts=${1:-3}
    local delay=${2:-2}
    local attempt=1
    
    while (( attempt <= max_attempts )); do
        if "${@:3}"; then
            return 0
        fi
        echo "尝试 $attempt/$max_attempts 失败，${delay}秒后重试..." >&2
        sleep "$delay"
        (( attempt++ ))
        (( delay *= 2 ))    # 指数退避
    done
    
    return 1
}

retry 5 3 curl -s https://api.example.com/health
```

### 编写健壮脚本：完整示例

```bash
#!/usr/bin/env bash
#
# backup.sh — 数据库备份脚本
# 用法: ./backup.sh [options]
#

set -euo pipefail

# === 配置 ===
readonly SCRIPT_NAME=$(basename "$0")
readonly DEFAULT_BACKUP_DIR="/var/backups/mysql"
readonly TIMESTAMP=$(date +%Y%m%d_%H%M%S)

# === 函数定义 ===

# 打印用法
usage() {
    cat <<EOF
用法: ${SCRIPT_NAME} [选项]

选项:
    -d DATABASE    要备份的数据库名（必填）
    -o OUTPUT_DIR  输出目录（默认: ${DEFAULT_BACKUP_DIR}）
    -c             对备份文件进行压缩
    -v             详细输出
    -h             显示此帮助
EOF
    exit 0
}

# 日志输出
log() {
    local level="$1"
    shift
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [${level}] $*"
}

# 错误退出
die() {
    log "ERROR" "$*"
    exit 1
}

# 清理临时文件
cleanup() {
    if [ -n "${TEMP_DIR:-}" && -d "$TEMP_DIR" ]( -n "${TEMP_DIR:-}" && -d "$TEMP_DIR" .md); then
        log "INFO" "清理临时目录: ${TEMP_DIR}"
        rm -rf "$TEMP_DIR"
    fi
}

# 备份数据库
backup_database() {
    local db="$1"
    local out_dir="$2"
    local compress="${3:-false}"
    local out_file="${out_dir}/${db}_${TIMESTAMP}.sql"
    
    log "INFO" "开始备份数据库: ${db}"
    
    # 使用 mysqldump 备份
    if ! mysqldump --single-transaction --routines --triggers \
        "$db" > "$out_file" 2>/dev/null; then
        die "数据库备份失败: ${db}"
    fi
    
    # 可选压缩
    if [ "$compress" = "true" ]( "$compress" = "true" .md); then
        log "INFO" "压缩备份文件..."
        gzip "$out_file"
        out_file="${out_file}.gz"
    fi
    
    local file_size
    file_size=$(stat -c%s "$out_file" 2>/dev/null || stat -f%z "$out_file" 2>/dev/null)
    log "INFO" "备份完成: ${out_file} (${file_size} bytes)"
}

# === 主流程 ===

main() {
    local database=""
    local output_dir="${DEFAULT_BACKUP_DIR}"
    local compress=false
    local verbose=false
    
    # 注册退出清理
    trap cleanup EXIT
    
    # 解析参数
    while getopts ":d:o:cvh" opt; do
        case $opt in
            d) database="$OPTARG" ;;
            o) output_dir="$OPTARG" ;;
            c) compress=true ;;
            v) verbose=true ;;
            h) usage ;;
            \?) die "无效选项: -$OPTARG；使用 -h 查看帮助" ;;
            :) die "选项 -$OPTARG 需要参数" ;;
        esac
    done
    
    # 验证参数
    if [ -z "$database" ]( -z "$database" .md); then
        die "数据库名是必填参数；使用 -h 查看帮助"
    fi
    
    # 创建输出目录
    if [ ! -d "$output_dir" ]( ! -d "$output_dir" .md); then
        mkdir -p "$output_dir" || die "无法创建输出目录: ${output_dir}"
    fi
    
    # 执行备份
    backup_database "$database" "$output_dir" "$compress"
    
    log "INFO" "备份脚本执行完毕"
}

# 进入主流程
main "$@"
```

> Linux 设备驱动是内核与硬件设备之间的桥梁，负责将硬件能力暴露给用户空间。理解驱动框架是内核开发和系统编程的核心技能。

## 核心概念

| 概念 | 说明 |
|------|------|
| 设备文件 | `/dev/` 下的特殊文件，通过文件操作访问硬件 |
| 主设备号 (major) | 标识设备驱动程序 |
| 次设备号 (minor) | 标识同一驱动管理的不同设备实例 |
| sysfs | `/sys/` 文件系统，导出设备/驱动拓扑信息 |
| 设备模型 | 基于 kobject 的统一设备/驱动/总线模型 |

## 内核模块 (Kernel Module)

### .ko 文件结构

内核模块是 `.ko`（kernel object）格式的 ELF 文件，与用户空间 ELF 类似但包含内核特有的 section：

| Section | 用途 |
|---------|------|
| `.modinfo` | 模块信息（作者、许可证、参数描述） |
| `.gnu.linkonce.this_module` | `struct module` 实例 |
| `.init.text` | `module_init()` 函数的代码 |
| `.exit.text` | `module_exit()` 函数的代码 |
| `__versions` | 符号版本信息（CRC） |

### 模块加载流程 (`insmod`)

```
insmod → 内核 sys_finit_module()
    → load_module(): 解析 ELF、分配内存、重定位符号
    → do_init_module(): 调用 module_init()
    → 模块状态变为 MODULE_STATE_LIVE
```

### 模块卸载流程 (`rmmod`)

```
rmmod → 内核 sys_delete_module()
    → 检查模块引用计数 (try_module_get/module_put)
    → 调用 module_exit()
    → 释放模块占用的内存
```

### 常用命令

| 命令 | 功能 |
|------|------|
| `insmod mod.ko` | 加载模块（需指定路径） |
| `rmmod mod` | 卸载模块 |
| `modprobe mod` | 自动加载模块及其依赖 |
| `lsmod` | 列出已加载模块 |
| `modinfo mod.ko` | 查看模块信息 |
| `depmod` | 生成模块依赖关系 |

### Hello World 内核模块

```c
#include <linux/init.h>
#include <linux/module.h>
#include <linux/kernel.h>

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("A simple Hello World kernel module");
MODULE_VERSION("1.0");

static int __init hello_init(void)
{
    pr_info("Hello, Kernel! Module loaded.\n");
    return 0; // 0 = success, negative = error
}

static void __exit hello_exit(void)
{
    pr_info("Goodbye, Kernel! Module unloaded.\n");
}

module_init(hello_init);
module_exit(hello_exit);
```

对应的 Makefile：

```makefile
obj-m += hello.o

all:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules

clean:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean
```

## 字符设备驱动

### 核心数据结构

```c
#include <linux/fs.h>
#include <linux/cdev.h>

// 字符设备描述符
struct cdev {
    struct kobject kobj;
    struct module *owner;
    const struct file_operations *ops;
    dev_t dev;          // 设备号 (major + minor)
    unsigned int count; // 管理的次设备号数量
};

// 文件操作表 - 连接 VFS 和驱动实现
struct file_operations {
    struct module *owner;
    loff_t  (*llseek)  (struct file *, loff_t, int);
    ssize_t (*read)    (struct file *, char __user *, size_t, loff_t *);
    ssize_t (*write)   (struct file *, const char __user *, size_t, loff_t *);
    long    (*unlocked_ioctl)(struct file *, unsigned int, unsigned long);
    int     (*open)    (struct inode *, struct file *);
    int     (*release) (struct inode *, struct file *);
    // ... 更多操作
};
```

### 主次设备号

* **主设备号**: 内核全局分配，标识驱动（`/proc/devices` 查看）
* **次设备号**: 由驱动自行管理，标识不同设备实例
* 静态分配: `register_chrdev_region(dev, count, name)`
* 动态分配: `alloc_chrdev_region(&dev, 0, count, name)`

### 注册流程

```
1. alloc_chrdev_region()     → 动态分配设备号
2. cdev_init(&cdev, &fops)   → 初始化 cdev
3. cdev_add(&cdev, dev, 1)   → 注册到内核
4. class_create() + device_create() → 自动创建设备文件 (/dev/xxx)
```

### 完整的字符设备驱动示例

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/device.h>

#define DEVICE_NAME "chardev"
#define BUFFER_SIZE 1024

static int major;
static struct cdev *chardev_cdev;
static struct class *chardev_class;
static char *device_buffer;

static int chardev_open(struct inode *inode, struct file *file)
{
    pr_info("chardev: device opened\n");
    return 0;
}

static int chardev_release(struct inode *inode, struct file *file)
{
    pr_info("chardev: device closed\n");
    return 0;
}

static ssize_t chardev_read(struct file *file, char __user *buf,
                            size_t count, loff_t *offset)
{
    size_t available = strlen(device_buffer);

    if (*offset >= available)
        return 0; // EOF

    if (count > available - *offset)
        count = available - *offset;

    if (copy_to_user(buf, device_buffer + *offset, count))
        return -EFAULT;

    *offset += count;
    pr_info("chardev: read %zu bytes\n", count);
    return count;
}

static ssize_t chardev_write(struct file *file, const char __user *buf,
                             size_t count, loff_t *offset)
{
    if (count > BUFFER_SIZE - 1)
        count = BUFFER_SIZE - 1;

    memset(device_buffer, 0, BUFFER_SIZE);
    if (copy_from_user(device_buffer, buf, count))
        return -EFAULT;

    device_buffer[count] = '\0';
    *offset = count;
    pr_info("chardev: wrote %zu bytes\n", count);
    return count;
}

static long chardev_ioctl(struct file *file, unsigned int cmd,
                          unsigned long arg)
{
    pr_info("chardev: ioctl cmd=%u\n", cmd);
    return 0;
}

static struct file_operations chardev_fops = {
    .owner          = THIS_MODULE,
    .open           = chardev_open,
    .release        = chardev_release,
    .read           = chardev_read,
    .write          = chardev_write,
    .unlocked_ioctl = chardev_ioctl,
};

static int __init chardev_init(void)
{
    dev_t dev;

    // 1. 动态分配设备号
    if (alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME) < 0) {
        pr_err("chardev: failed to allocate device number\n");
        return -1;
    }
    major = MAJOR(dev);

    // 2. 初始化 cdev
    chardev_cdev = cdev_alloc();
    if (!chardev_cdev) {
        unregister_chrdev_region(dev, 1);
        return -ENOMEM;
    }
    cdev_init(chardev_cdev, &chardev_fops);
    chardev_cdev->owner = THIS_MODULE;

    // 3. 注册到内核
    if (cdev_add(chardev_cdev, dev, 1) < 0) {
        pr_err("chardev: cdev_add failed\n");
        cdev_del(chardev_cdev);
        unregister_chrdev_region(dev, 1);
        return -1;
    }

    // 4. 在 /dev/ 下创建设备文件
    chardev_class = class_create(THIS_MODULE, DEVICE_NAME);
    device_create(chardev_class, NULL, dev, NULL, DEVICE_NAME);

    // 5. 分配缓冲区
    device_buffer = kmalloc(BUFFER_SIZE, GFP_KERNEL);
    if (!device_buffer) {
        pr_err("chardev: kmalloc failed\n");
        cdev_del(chardev_cdev);
        unregister_chrdev_region(dev, 1);
        return -ENOMEM;
    }
    memset(device_buffer, 0, BUFFER_SIZE);

    pr_info("chardev: loaded, major=%d\n", major);
    return 0;
}

static void __exit chardev_exit(void)
{
    dev_t dev = MKDEV(major, 0);

    device_destroy(chardev_class, dev);
    class_destroy(chardev_class);
    cdev_del(chardev_cdev);
    unregister_chrdev_region(dev, 1);
    kfree(device_buffer);

    pr_info("chardev: unloaded\n");
}

module_init(chardev_init);
module_exit(chardev_exit);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("A simple character device driver");
MODULE_AUTHOR("Your Name");
```

### 测试字符设备

```bash
# 加载模块后，/dev/chardev 自动创建
echo "Hello Device" > /dev/chardev   # 触发 write
cat /dev/chardev                     # 触发 read
```

## 块设备驱动

块设备与字符设备的核心区别：

| 特性 | 字符设备 | 块设备 |
|------|----------|--------|
| 访问方式 | 字节流 (stream) | 块 (block, 通常 512B/4KB) |
| 随机访问 | 支持 seek | 天然支持 (按块寻址) |
| 缓冲 | 无内核缓冲 | 通过 Page Cache + 请求队列 |
| 主要接口 | `file_operations` | `block_device_operations` |
| 示例 | 串口、GPIO | 硬盘、SSD |

块设备关键结构：

```c
// 块设备操作接口
struct block_device_operations {
    int (*open)(struct block_device *, fmode_t);
    void (*release)(struct gendisk *, fmode_t);
    int (*ioctl)(struct block_device *, fmode_t, unsigned, unsigned long);
    int (*direct_access)(struct block_device *, sector_t, void **, unsigned long *);
    struct module *owner;
};

// 通用磁盘描述
struct gendisk {
    int major;                          // 主设备号
    int first_minor;                    // 起始次设备号
    int minors;                         // 次设备号数量
    char disk_name[DISK_NAME_LEN];      // 名称 (如 "sda")
    struct block_device_operations *fops;
    struct request_queue *queue;        // 请求队列
    void *private_data;
};
```

## DMA 与 MMIO

对于 PCIe 等高速设备，驱动需要理解两种硬件交互方式：

### MMIO (Memory-Mapped IO)

* 设备的控制/状态寄存器映射到 CPU 的物理地址空间
* 使用 `ioremap()` 或 `ioremap_wc()` 建立内核虚拟地址映射
* 通过 `readl()/writel()` 等访问寄存器（保证顺序性）

```c
// PCIe BAR 映射示例
void __iomem *bar0 = pci_iomap(pdev, 0, pci_resource_len(pdev, 0));
writel(value, bar0 + REG_OFFSET);  // 写寄存器
u32 val = readl(bar0 + REG_STATUS); // 读状态
```

### DMA (Direct Memory Access)

* 设备直接读写主机内存，无需 CPU 逐字搬运
* 驱动使用 DMA API 分配一致性/流式 DMA 缓冲区

```c
// DMA 分配示例
dma_addr_t dma_handle;
void *cpu_addr = dma_alloc_coherent(dev, size, &dma_handle, GFP_KERNEL);

// 告诉设备 DMA 地址，设备直接读写该内存
writel(dma_handle, bar0 + DMA_ADDR_REG);
writel(size, bar0 + DMA_SIZE_REG);
writel(1, bar0 + DMA_START_REG); // 启动 DMA
```

| DMA API | 用途 |
|---------|------|
| `dma_alloc_coherent()` | 分配一致性 DMA 缓冲区（cache 一致） |
| `dma_map_single()` | 映射流式 DMA 缓冲区 |
| `dma_unmap_single()` | 解除流式 DMA 映射 |
| `dma_alloc_noncoherent()` | 分配非一致性缓冲区（需手动 flush） |

## 设备文件创建机制

### mknod (手动创建)

```bash
# 语法: mknod <name> <type> <major> <minor>
mknod /dev/mydevice c 240 0   # 字符设备
mknod /dev/myblock b 8 0      # 块设备
```

### udev (自动创建)

现代 Linux 使用 `udev`（基于 `devtmpfs` + `udevd`）自动管理设备文件：

```
内核检测到新设备 → 发送 uevent 到 udevd
    → udevd 匹配规则 (/etc/udev/rules.d/)
    → 在 /dev/ 下创建设备文件
    → 加载固件、设置权限、触发模块加载
```

驱动的 `class_create()` + `device_create()` 会触发内核发送 uevent，从而让 udev 自动创建设备文件。

## sysfs (设备模型)

sysfs 挂载在 `/sys/`，以目录树形式导出内核对象：

```
/sys/
├── bus/        → 总线 (pci, usb, platform...)
├── class/      → 设备类 (net, input, tty, misc...)
├── devices/    → 所有设备
└── module/     → 已加载的内核模块
```

驱动可以通过 `sysfs_create_group()` 创建自定义属性文件：

```c
static ssize_t myattr_show(struct kobject *kobj, struct kobj_attribute *attr, char *buf)
{
    return sprintf(buf, "%d\n", my_value);
}

static struct kobj_attribute myattr = __ATTR_RO(myattr);

static struct attribute *attrs[] = {
    &myattr.attr,
    NULL,
};

static struct attribute_group myattr_group = {
    .attrs = attrs,
};

sysfs_create_group(&pdev->dev.kobj, &myattr_group);
```

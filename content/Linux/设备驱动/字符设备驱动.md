> 字符设备是 Linux 中最常见的设备类型，以字节流方式访问，通过 `file_operations` 结构体向 VFS 层暴露 `read`/`write`/`ioctl` 等标准文件操作接口。

## 核心数据结构

```c
#include <linux/fs.h>
#include <linux/cdev.h>

// 字符设备描述符
struct cdev {
    struct kobject kobj;
    struct module *owner;
    const struct file_operations *ops;
    dev_t dev;          // 设备号 (major + minor)
    unsigned int count; // 管理的次设备号数量
};

// 文件操作表 - 连接 VFS 和驱动实现
struct file_operations {
    struct module *owner;
    loff_t  (*llseek)  (struct file *, loff_t, int);
    ssize_t (*read)    (struct file *, char __user *, size_t, loff_t *);
    ssize_t (*write)   (struct file *, const char __user *, size_t, loff_t *);
    long    (*unlocked_ioctl)(struct file *, unsigned int, unsigned long);
    int     (*open)    (struct inode *, struct file *);
    int     (*release) (struct inode *, struct file *);
    // ... 更多操作
};
```

## 主次设备号

- **主设备号**: 内核全局分配，标识驱动（`/proc/devices` 查看）
- **次设备号**: 由驱动自行管理，标识不同设备实例
- 静态分配: `register_chrdev_region(dev, count, name)`
- 动态分配: `alloc_chrdev_region(&dev, 0, count, name)`

## 注册流程

```
1. alloc_chrdev_region()     → 动态分配设备号
2. cdev_init(&cdev, &fops)   → 初始化 cdev
3. cdev_add(&cdev, dev, 1)   → 注册到内核
4. class_create() + device_create() → 自动创建设备文件 (/dev/xxx)
```

## 完整的字符设备驱动示例

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/device.h>

#define DEVICE_NAME "chardev"
#define BUFFER_SIZE 1024

static int major;
static struct cdev *chardev_cdev;
static struct class *chardev_class;
static char *device_buffer;

static int chardev_open(struct inode *inode, struct file *file)
{
    pr_info("chardev: device opened\n");
    return 0;
}

static int chardev_release(struct inode *inode, struct file *file)
{
    pr_info("chardev: device closed\n");
    return 0;
}

static ssize_t chardev_read(struct file *file, char __user *buf,
                            size_t count, loff_t *offset)
{
    size_t available = strlen(device_buffer);

    if (*offset >= available)
        return 0; // EOF

    if (count > available - *offset)
        count = available - *offset;

    if (copy_to_user(buf, device_buffer + *offset, count))
        return -EFAULT;

    *offset += count;
    pr_info("chardev: read %zu bytes\n", count);
    return count;
}

static ssize_t chardev_write(struct file *file, const char __user *buf,
                             size_t count, loff_t *offset)
{
    if (count > BUFFER_SIZE - 1)
        count = BUFFER_SIZE - 1;

    memset(device_buffer, 0, BUFFER_SIZE);
    if (copy_from_user(device_buffer, buf, count))
        return -EFAULT;

    device_buffer[count] = '\0';
    *offset = count;
    pr_info("chardev: wrote %zu bytes\n", count);
    return count;
}

static long chardev_ioctl(struct file *file, unsigned int cmd,
                          unsigned long arg)
{
    pr_info("chardev: ioctl cmd=%u\n", cmd);
    return 0;
}

static struct file_operations chardev_fops = {
    .owner          = THIS_MODULE,
    .open           = chardev_open,
    .release        = chardev_release,
    .read           = chardev_read,
    .write          = chardev_write,
    .unlocked_ioctl = chardev_ioctl,
};

static int __init chardev_init(void)
{
    dev_t dev;

    // 1. 动态分配设备号
    if (alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME) < 0) {
        pr_err("chardev: failed to allocate device number\n");
        return -1;
    }
    major = MAJOR(dev);

    // 2. 初始化 cdev
    chardev_cdev = cdev_alloc();
    if (!chardev_cdev) {
        unregister_chrdev_region(dev, 1);
        return -ENOMEM;
    }
    cdev_init(chardev_cdev, &chardev_fops);
    chardev_cdev->owner = THIS_MODULE;

    // 3. 注册到内核
    if (cdev_add(chardev_cdev, dev, 1) < 0) {
        pr_err("chardev: cdev_add failed\n");
        cdev_del(chardev_cdev);
        unregister_chrdev_region(dev, 1);
        return -1;
    }

    // 4. 在 /dev/ 下创建设备文件
    chardev_class = class_create(THIS_MODULE, DEVICE_NAME);
    device_create(chardev_class, NULL, dev, NULL, DEVICE_NAME);

    // 5. 分配缓冲区
    device_buffer = kmalloc(BUFFER_SIZE, GFP_KERNEL);
    if (!device_buffer) {
        pr_err("chardev: kmalloc failed\n");
        cdev_del(chardev_cdev);
        unregister_chrdev_region(dev, 1);
        return -ENOMEM;
    }
    memset(device_buffer, 0, BUFFER_SIZE);

    pr_info("chardev: loaded, major=%d\n", major);
    return 0;
}

static void __exit chardev_exit(void)
{
    dev_t dev = MKDEV(major, 0);

    device_destroy(chardev_class, dev);
    class_destroy(chardev_class);
    cdev_del(chardev_cdev);
    unregister_chrdev_region(dev, 1);
    kfree(device_buffer);

    pr_info("chardev: unloaded\n");
}

module_init(chardev_init);
module_exit(chardev_exit);

MODULE_LICENSE("GPL");
MODULE_DESCRIPTION("A simple character device driver");
MODULE_AUTHOR("Your Name");
```

## 测试字符设备

```bash
# 加载模块后，/dev/chardev 自动创建
echo "Hello Device" > /dev/chardev   # 触发 write
cat /dev/chardev                     # 触发 read
```

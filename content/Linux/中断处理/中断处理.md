> 中断处理是 Linux 内核响应硬件事件的机制。为了在中断屏蔽时间最短的同时完成必要工作，内核将中断处理分为顶半部 (Top Half) 和底半部 (Bottom Half)，分别运行在中断上下文和软中断/进程上下文中。

## 核心概念

| 概念 | 说明 |
|------|------|
| 中断上下文 | 中断处理程序执行环境，不可睡眠，不可触发缺页 |
| 进程上下文 | 普通内核代码执行环境，可以睡眠 |
| IRQ 号 | 内核分配给每个中断源的逻辑编号 |
| 顶半部/硬中断 (HardIRQ) | 注册后立即执行，全部或部分关闭中断 |
| 底半部 (Bottom Half) | 延迟执行中断处理，分为 softirq/tasklet/workqueue |

## 整体架构

```
硬件设备产生中断
    │
    ▼
中断控制器 (GIC/APIC/IOAPIC)
    │
    ▼
CPU 接收到中断信号
    │  保存上下文、关中断、进入 IDT 表项
    ▼
do_IRQ() / do_interrupt()
    │  查找 irq_desc
    ▼
generic_handle_irq()
    │
    ▼
irq_desc::handle_irq()
    │
    ├── handle_fasteoi_irq    (MSI/MSI-X, 写 EOI 通知控制器)
    ├── handle_edge_irq       (边沿触发, 自动 ACK + 重入检测)
    └── handle_level_irq      (电平触发, 先 Mask 再 ACK)
    │
    ▼
action->handler (request_irq 注册的 ISR)
    │  [Top Half: 快速执行，ACK 设备、保存关键数据]
    │  ⚠️ 不可睡眠，中断关闭
    ▼
Bottom Half (延迟处理)
    ├── softirq  (固定 10 个，同一类型可多 CPU 并行)
    ├── tasklet  (动态分配，同一 tasklet 串行执行)
    ├── workqueue (进程上下文，可以睡眠)
    └── threaded IRQ (独立内核线程，可睡眠)
```

## 顶半部 (HardIRQ)

### 中断注册

```c
#include <linux/interrupt.h>

// request_irq 原型
int request_irq(unsigned int irq,                // 中断号
                irq_handler_t handler,           // 处理函数
                unsigned long flags,             // 标志位
                const char *name,                // 名称 (显示在 /proc/interrupts)
                void *dev);                      // 设备标识 (用于共享中断)

// 返回值: 0 成功，负值错误码
```

### 中断标志位

| 标志 | 含义 |
|------|------|
| `IRQF_SHARED` | 允许多个设备共享同一个 IRQ 线 |
| `IRQF_TRIGGER_RISING` | 上升沿触发 |
| `IRQF_TRIGGER_FALLING` | 下降沿触发 |
| `IRQF_TRIGGER_HIGH` | 高电平触发 |
| `IRQF_TRIGGER_LOW` | 低电平触发 |
| `IRQF_ONESHOT` | 执行期间禁止中断自动重新使能 |

### 处理函数约束

```c
// handler 必须运行在中断上下文
static irqreturn_t my_interrupt_handler(int irq, void *dev_id)
{
    // ✅ 允许的操作:
    //   - 读写硬件寄存器 (readl/writel)
    //   - 操作硬件数据结构 (指针、整数)
    //   - 触发底半部 (tasklet_schedule / workqueue)

    // ❌ 禁止的操作:
    //   - 睡眠 (msleep, wait_event)
    //   - 获取互斥锁 (mutex_lock)
    //   - 访问用户空间 (copy_from_user / copy_to_user)
    //   - 执行可能会阻塞的函数 (kmalloc(GFP_KERNEL))

    // 返回值:
    return IRQ_HANDLED;      // 该中断由本 handler 处理
    // return IRQ_NONE;      // 不是本设备的中断 (共享中断用)
}
```

### request_irq 示例

```c
#include <linux/module.h>
#include <linux/interrupt.h>
#include <linux/io.h>
#include <linux/device.h>

#define MY_IRQ_NUM  42
#define DEV_NAME    "my_device"

static unsigned long *mmio_base;

static irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    struct device *dev = (struct device *)dev_id;
    u32 status;

    // 读取设备状态寄存器确认中断源
    status = readl(mmio_base + STATUS_REG);

    if (!(status & INT_PENDING))
        return IRQ_NONE; // 不是本设备的中断

    // 清除中断标志 (大多数设备要求显式 ACK)
    writel(status | INT_CLEAR, mmio_base + STATUS_REG);

    pr_info("IRQ %d handled, status=0x%x\n", irq, status);

    // 触发底半部处理 (tasklet 或 workqueue)

    return IRQ_HANDLED;
}

static int __init my_device_init(void)
{
    int ret;

    // 假设已映射 mmio_base
    // mmio_base = ioremap(...);

    ret = request_irq(MY_IRQ_NUM, my_irq_handler,
                      IRQF_SHARED | IRQF_TRIGGER_RISING,
                      DEV_NAME, my_dev);
    if (ret) {
        pr_err("request_irq failed: %d\n", ret);
        return ret;
    }
    return 0;
}

static void __exit my_device_exit(void)
{
    free_irq(MY_IRQ_NUM, my_dev); // 必须传入与 request_irq 相同的 dev_id
}

module_init(my_device_init);
module_exit(my_device_exit);
```

## 底半部机制对比

| 机制 | 运行上下文 | 可睡眠 | 可并行 | 固定/动态 | 使用场景 |
|------|-----------|--------|--------|-----------|----------|
| softirq | 中断上下文 | ❌ | ✅ (同一类型可多核同时执行) | 固定 10 个 | 网络收发、时钟 |
| tasklet | 中断上下文 (softirq 中执行) | ❌ | ❌ (同一 tasklet 串行) | 动态分配 | 一般设备驱动 |
| workqueue | 进程上下文 | ✅ | ✅ | 动态分配 | 需要睡眠的延迟工作 |

### softirq (软中断)

softirq 编译时静态分配，内核预定义了 10 个优先级：

| 软中断索引 | 名称 | 用途 |
|-----------|------|------|
| 0 | `HI_SOFTIRQ` | 高优先级 tasklet |
| 1 | `TIMER_SOFTIRQ` | 定时器 |
| 2 | `NET_TX_SOFTIRQ` | 网络发送 |
| 3 | `NET_RX_SOFTIRQ` | 网络接收 |
| 4 | `BLOCK_SOFTIRQ` | 块设备 |
| 6 | `TASKLET_SOFTIRQ` | 普通 tasklet |
| 7 | `SCHED_SOFTIRQ` | 调度器 |
| 9 | `RCU_SOFTIRQ` | RCU 处理 |

软中断在以下时机被处理：

| 触发点 | 说明 |
|--------|------|
| `do_IRQ()` 返回前 | 硬中断结束后立即检查 |
| `local_bh_enable()` | 底半部重新使能时 |
| `ksoftirqd` 内核线程 | 当软中断累积过多时，由 `ksoftirqd/N` 处理 |

### tasklet (小任务)

tasklet 基于 softirq 实现，属于 `HI_SOFTIRQ` 或 `TASKLET_SOFTIRQ`。驱动通常使用 tasklet 而非直接使用 softirq。

```c
#include <linux/interrupt.h>

// 定义 tasklet 函数
void my_tasklet_handler(unsigned long data);

// 声明并初始化 tasklet
DECLARE_TASKLET(my_tasklet, my_tasklet_handler, 0);
// 或者运行时初始化
// tasklet_init(&my_tasklet, my_tasklet_handler, 0);

// 在中断顶半部中调度
void my_irq_handler(int irq, void *dev_id)
{
    // ... 快速处理硬件 ...
    tasklet_schedule(&my_tasklet); // 标记 tasklet 为待运行
    // tasklet_hi_schedule()  // 高优先级版本
}

void my_tasklet_handler(unsigned long data)
{
    // tasklet 运行在软中断上下文
    // 不可睡眠，但本 tasklet 不会和其他 CPU 上的本 tasklet 并发
    pr_info("tasklet is running\n");
}

// 清理
void cleanup(void)
{
    tasklet_kill(&my_tasklet); // 确保 tasklet 不会在模块卸载后运行
}
```

### workqueue (工作队列)

workqueue 运行在进程上下文（内核线程），可以睡眠，适合执行需要阻塞的操作。

```c
#include <linux/workqueue.h>

// 1. 使用系统默认 workqueue (system_wq)
static struct work_struct my_work;

void my_work_handler(struct work_struct *work)
{
    // 运行在进程上下文 (kworker 线程)
    // ✅ 可以睡眠、获取 mutex、访问用户空间
    msleep(100);
    pr_info("work handler running in process context\n");
}

void my_irq_handler(int irq, void *dev_id)
{
    schedule_work(&my_work); // 将 work 提交到 system_wq
}

// 初始化
INIT_WORK(&my_work, my_work_handler);


// 2. 创建自定义 workqueue (专用 kworker 线程)
static struct workqueue_struct *my_wq;
static struct work_struct my_custom_work;

void my_custom_work_handler(struct work_struct *work)
{
    /* ... */
}

// 创建
my_wq = create_singlethread_workqueue("my_wq");
INIT_WORK(&my_custom_work, my_custom_work_handler);

// 提交到自定义 wq
queue_work(my_wq, &my_custom_work);

// 销毁
destroy_workqueue(my_wq);
```

| Workqueue API | 说明 |
|---------------|------|
| `schedule_work()` | 提交到系统默认 workqueue `system_wq` |
| `schedule_delayed_work()` | 延迟一段时间后执行 |
| `flush_work()` | 等待 work 执行完成 |
| `cancel_work_sync()` | 取消 work 并等待完成 |
| `create_singlethread_workqueue()` | 创建单线程 workqueue |
| `create_workqueue()` | 创建 per-CPU 多线程 workqueue |

## Linux 中断子系统架构

```
硬件中断
    │
    ▼
irq_chip (硬件层抽象)
    │  → 操作中断控制器寄存器 (mask/unmask/ack/eoi)
    ▼
irq_domain (映射层)
    │  → HW IRQ → Linux IRQ 号转换
    ▼
irq_desc (核心数据结构)
    │  → 每个 Linux IRQ 对应一个 irq_desc
    │  → 包含 action 链表 (request_irq 注册的 handler)
    ▼
handle_irq() (高层中断处理)
    │  → 调用 irq_desc->handle_irq()
    │  → 遍历 action 链表执行 handler
    ▼
Bottom Half (softirq / tasklet / workqueue)
```

### irq_desc

每个 Linux IRQ 号对应一个 `irq_desc`，是中断处理的中心数据结构：

```c
struct irq_desc {
    struct irq_common_data  irq_common_data;
    struct irq_data         irq_data;
    unsigned int __percpu   *kstat_irqs;       // 每个 CPU 的中断统计
    irq_flow_handler_t      handle_irq;        // 高层处理函数
    struct irqaction        *action;           // 注册的 handler 链表
    unsigned int            status_use_accessors;
    const char              *name;
    // ...
};
```

### irq_chip (中断控制器抽象)

```c
struct irq_chip {
    const char      *name;
    void (*irq_mask)(struct irq_data *data);      // 屏蔽中断
    void (*irq_unmask)(struct irq_data *data);    // 使能中断
    void (*irq_ack)(struct irq_data *data);       // 应答中断
    void (*irq_eoi)(struct irq_data *data);       // 中断结束
    void (*irq_set_type)(struct irq_data *data, unsigned int flow_type); // 设置触发方式
    // ...
};
```

### irq_domain (中断映射)

`irq_domain` 负责将硬件中断号（如 GIC 的 SPI ID）映射为 Linux IRQ 号：

```c
struct irq_domain *domain;

// 创建映射域
domain = irq_domain_add_linear(node, num_hw_irqs, &ops, host_data);

// 将硬件 IRQ 映射到 Linux IRQ
unsigned int virq = irq_create_mapping(domain, hwirq);

// 直接查找映射
unsigned int virq = irq_find_mapping(domain, hwirq);
```

## THREADED IRQ (线程化中断)

现代内核支持线程化中断，将中断处理作为内核线程运行，允许睡眠：

```c
// request_threaded_irq 替代 request_irq
int request_threaded_irq(unsigned int irq,
                         irq_handler_t handler,       // 顶半部 (可为 NULL)
                         irq_handler_t thread_fn,     // 线程化处理函数 (可睡眠)
                         unsigned long flags,
                         const char *name,
                         void *dev);

// 示例: 顶半部极简，所有工作在线程中完成
static irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    // 快速确认设备，触发线程
    return IRQ_WAKE_THREAD; // 唤醒中断线程运行 thread_fn
}

static irqreturn_t my_irq_thread_fn(int irq, void *dev_id)
{
    // 运行在进程上下文，可以睡眠
    msleep_interruptible(10);
    // ... 耗时的设备操作 ...
    return IRQ_HANDLED;
}

request_threaded_irq(irq_num, my_irq_handler, my_irq_thread_fn,
                     IRQF_TRIGGER_RISING, "my_dev", dev);
```

## 中断延迟与实时性

中断延迟定义为从硬件产生中断信号到 CPU 开始执行 ISR 的时间：

```
硬件中断信号
  → PCIe/系统总线延迟（纳秒级）
    → CPU 完成当前指令（纳秒级）
      → CPU 查找 IDT / 跳转到中断入口（几十纳秒）
        → 关中断、保存上下文（几百纳秒）
          → IRQ 分发 → 执行 ISR（取决于 ISR 耗时）
```

| 因素 | 对延迟的影响 |
|------|-------------|
| 中断屏蔽（`local_irq_disable` / spinlock） | 最长延迟来源：若 spinlock 持有者被抢占，延迟不可控 |
| 中断共享（IRQF_SHARED） | 多个设备共享同一 IRQ 线，ISR 链遍历增加延迟 |
| MSI/MSI-X vs IO-APIC | MSI/MSI-X 绕过 I/O APIC，延迟更低 |
| CPU 缓存命中率 | 中断处理代码和数据在缓存中时延迟更低 |
| 中断亲和性设置不当 | 中断分配到繁忙核心，排队等待 |

### PREEMPT_RT

PREEMPT_RT（Real-Time Linux）补丁系列将 Linux 转变为硬实时系统：

| 特性 | 标准内核 | PREEMPT_RT |
|------|---------|------------|
| spinlock | 禁止抢占 + 禁止中断 | 可睡眠（rt-mutex 实现） |
| interrupt handler | 在中断上下文中运行 | 全部在进程上下文中运行（threaded IRQ） |
| `local_irq_disable` | 真正关中断 | 仅标记，不影响硬件中断 |
| 最大延迟 | 毫秒~数十毫秒（不可预测） | 微秒~几十微秒（可预测） |

自 Linux 5.x 起，PREEMPT_RT 的大部分功能已合入主线，通过 `CONFIG_PREEMPT_RT` 启用。

## 调试工具

```bash
# 查看所有中断分布和类型
cat /proc/interrupts

# 查看软中断统计
cat /proc/softirqs

# 查看每个中断的亲和性设置
ls /proc/irq/
cat /proc/irq/N/smp_affinity

# 查看中断触发计数（每秒刷新）
watch -n1 cat /proc/interrupts

# 查看 Local APIC 状态 (x86)
cat /proc/cpuinfo | grep apicid

# 查看 MSI/MSI-X 配置
lspci -vvv -s 01:00.0 | grep -A10 "MSI"

# 测量中断延迟（需要 cyclictest 工具）
cyclictest --mlockall --smp --priority=80 --interval=200 --distance=0

# 查看 irqbalance 状态
systemctl status irqbalance
```

> 内核同步是 Linux 内核中保护共享数据免受并发访问的机制集合。内核开发者需要面对来自多个 CPU、中断处理程序、软中断以及进程上下文的并发访问，选择合适的同步原语是保证正确性和性能的关键。

## 核心概念

| 概念 | 说明 |
|------|------|
| 临界区 | 访问共享资源的代码段，需要互斥保护 |
| 竞态条件 | 多个执行流对共享资源的非原子操作导致的结果不确定 |
| 死锁 | 两个或多个执行流互相等待对方持有的锁，永远阻塞 |
| 锁顺序 | 获取多个锁时必须按固定顺序，避免死锁 |
| 可重入 | 同一执行流能否重复获取同一把锁 |

## 同步机制全景对比

| 机制 | 上下文 | 可睡眠 | 自旋等待 | 重入 | 适用场景 |
|------|--------|--------|----------|------|----------|
| spinlock | 任意 | ❌ | ✅ (忙等) | ❌ (不可重入) | 中断/软中断保护的短临界区 |
| raw_spinlock | 任意 (更早期) | ❌ | ✅ | ❌ | 实时内核、早期启动代码 |
| mutex | 进程 | ✅ | ❌ | ❌ | 长临界区、需要睡眠 |
| semaphore | 进程 | ✅ | ❌ | ✅ (计数) | 资源计数 (有限数量) |
| rwlock | 任意 | ❌ | ✅ | ❌ | 读多写少、短临界区 |
| rwsem | 进程 | ✅ | ❌ | ❌ | 读多写少、长临界区 |
| RCU | 任意 | 读端: ❌ / 写端: ✅ | ❌ | ✅ | 读极多写极少、链表遍历 |
| atomic_t | 任意 | ❌ | ❌ (无锁) | ✅ | 简单计数器、标志位 |

## Spinlock (自旋锁)

自旋锁是最底层的锁机制：获取不到锁时原地循环（自旋），不放弃 CPU。

### 基本 API

```c
#include <linux/spinlock.h>

DEFINE_SPINLOCK(my_lock);  // 静态定义

// 基本使用
spin_lock(&my_lock);       // 自旋等待
// ... 临界区 ...
spin_unlock(&my_lock);

// 在中断上下文中使用时必须禁用本地中断
unsigned long flags;
spin_lock_irqsave(&my_lock, flags);   // 保存当前中断状态 + 关中断 + 获取锁
// ... 临界区 ...
spin_unlock_irqrestore(&my_lock, flags);  // 释放锁 + 恢复中断状态

// spin_lock_irqsave/spin_unlock_irqrestore 是最安全的变体
// 适合在中断处理程序 + 进程上下文混合访问的场景
```

### 变体对比

| API | 关闭本地中断 | 保存标志 | 适用场景 |
|-----|-------------|---------|----------|
| `spin_lock()` | ❌ | ❌ | 只有进程上下文竞争 |
| `spin_lock_irq()` | ✅ | ❌ | 进程 + 中断上下文，但已知中断状态 |
| `spin_lock_irqsave()` | ✅ | ✅ | 进程 + 中断上下文，推荐首选 |
| `spin_lock_bh()` | 关闭软中断 | ❌ | 进程 + 软中断上下文 |

### IRQ 中使用 spinlock 示例

```c
static DEFINE_SPINLOCK(dev_lock);
static struct device_data shared_data;

// 进程上下文
ssize_t dev_write(struct file *file, const char __user *buf,
                  size_t count, loff_t *offset)
{
    unsigned long flags;

    spin_lock_irqsave(&dev_lock, flags);
    // 修改共享数据
    shared_data.value = some_value;
    spin_unlock_irqrestore(&dev_lock, flags);

    return count;
}

// 中断上下文 - 也访问 shared_data
static irqreturn_t dev_irq_handler(int irq, void *dev_id)
{
    unsigned long flags;

    // 注意: 如果确定此 handler 执行时本地中断已关闭，也可用 spin_lock()
    // 但使用 _irqsave 更安全
    spin_lock_irqsave(&dev_lock, flags);
    // 读取/修改 shared_data
    shared_data.counter++;
    spin_unlock_irqrestore(&dev_lock, flags);

    return IRQ_HANDLED;
}
```

### raw_spinlock

`raw_spinlock` 是 spinlock 在 PREEMPT_RT 内核中的不变底层实现。在非 RT 内核中 `spinlock` 就是 `raw_spinlock`，但在 RT 内核中，`spinlock` 可被优先级继承的 `rt_mutex` 替代，而 `raw_spinlock` 始终是自旋锁。中断处理程序中的关键路径必须使用 `raw_spinlock`。

## Mutex (互斥锁)

Mutex 是进程上下文中的睡眠锁，无法获取锁时进程进入睡眠状态。

### 基本 API

```c
#include <linux/mutex.h>

DEFINE_MUTEX(my_mutex);  // 静态定义

// 使用
mutex_lock(&my_mutex);   // 获取锁，可能睡眠
// ... 临界区 ...
mutex_unlock(&my_mutex);

// 非阻塞尝试
if (mutex_trylock(&my_mutex)) { // 立即返回
    // ... 获取成功 ...
    mutex_unlock(&my_mutex);
} else {
    // ... 获取失败 ...
}
```

### Mutex 特性

| 特性 | 说明 |
|------|------|
| 所有者追踪 | `mutex_lock()` 记录当前所有者，用于调试 |
| 优先级继承 | 防止优先级反转 |
| 可递归 | ❌ 同一线程不能重复获取同一 mutex |
| 在中断上下文使用 | ❌ 会触发 `might_sleep()` 警告 |
| 约束 | 只能在进程上下文使用 |

### Mutex 在进程上下文示例

```c
#include <linux/mutex.h>
#include <linux/slab.h>
#include <linux/fs.h>
#include <linux/uaccess.h>

static DEFINE_MUTEX(device_mutex);
static char *device_buffer;
static size_t buffer_size = 1024;

ssize_t device_write(struct file *file, const char __user *buf,
                     size_t count, loff_t *offset)
{
    ssize_t ret;

    if (mutex_lock_interruptible(&device_mutex))  // 可被信号打断
        return -ERESTARTSYS;

    if (count > buffer_size) {
        ret = -ENOSPC;
        goto out;
    }

    if (copy_from_user(device_buffer, buf, count)) {
        ret = -EFAULT;
        goto out;
    }

    *offset = count;
    ret = count;

out:
    mutex_unlock(&device_mutex);
    return ret;
}
```

## RCU (Read-Copy-Update)

RCU 是一种适用于**读多写少**场景的无锁同步机制。读者不需要获取锁，写者通过发布/订阅和宽限期（grace period）实现安全更新。

### 核心思想

```
读者: rcu_read_lock() → 读取数据 → rcu_read_unlock()
写者: 1. 分配新数据结构并复制旧数据
     2. 更新指针 (rcu_assign_pointer)
     3. 等待所有读者完成 (synchronize_rcu / call_rcu)
     4. 释放旧数据
```

### RCU API

| API | 说明 |
|-----|------|
| `rcu_read_lock()` | 进入 RCU 读端临界区（不允许睡眠、阻塞） |
| `rcu_read_unlock()` | 退出 RCU 读端临界区 |
| `rcu_dereference()` | 读取受 RCU 保护的指针 |
| `rcu_assign_pointer()` | 发布更新后的指针 |
| `synchronize_rcu()` | 等待宽限期结束（阻塞，可睡眠） |
| `call_rcu()` | 注册回调，宽限期结束后执行（非阻塞） |
| `rcu_barrier()` | 等待所有已注册的 call_rcu 回调执行完毕 |

### RCU 示例: 保护只读链表

```c
#include <linux/rcupdate.h>
#include <linux/slab.h>

struct my_data {
    struct list_head list;
    int key;
    int value;
    struct rcu_head rcu_head; // 用于 call_rcu 延迟释放
};

static LIST_HEAD(data_list);

// 读者 — 无需锁
void read_entries(void)
{
    struct my_data *entry;

    rcu_read_lock();
    list_for_each_entry_rcu(entry, &data_list, list) {
        // 读取 entry->key, entry->value
        pr_info("key=%d, value=%d\n", entry->key, entry->value);
    }
    rcu_read_unlock();
}

// 写者 — 插入新节点
void add_entry(int key, int value)
{
    struct my_data *new_entry = kmalloc(sizeof(*new_entry), GFP_KERNEL);
    new_entry->key = key;
    new_entry->value = value;

    rcu_read_lock();      // 保护 list_add_rcu 的内部一致性
    list_add_rcu(&new_entry->list, &data_list);
    rcu_read_unlock();
}

// 写者 — 安全删除节点
static void free_entry_rcu(struct rcu_head *head)
{
    struct my_data *entry = container_of(head, struct my_data, rcu_head);
    kfree(entry);
}

void remove_entry(int key)
{
    struct my_data *entry;

    // 不允许此处进行 RCU 读端临界区
    rcu_read_lock();
    list_for_each_entry_rcu(entry, &data_list, list) {
        if (entry->key == key) {
            list_del_rcu(&entry->list);
            rcu_read_unlock();
            // 等待所有读者退出后释放
            call_rcu(&entry->rcu_head, free_entry_rcu);
            return;
        }
    }
    rcu_read_unlock();
}
```

## 原子操作 (Atomic Operations)

对于简单的整数操作，原子操作比锁更轻量。

```c
#include <linux/types.h>

atomic_t counter;    // 32 位原子变量
atomic64_t big;     // 64 位原子变量

// 初始化
atomic_set(&counter, 0);
// 或 ATOMIC_INIT(0);

// 操作
atomic_inc(&counter);                // counter++
atomic_dec(&counter);                // counter--
atomic_add(10, &counter);            // counter += 10
atomic_sub(5, &counter);             // counter -= 5
int val = atomic_read(&counter);     // 读取

// 原子操作并测试结果
int old = atomic_fetch_add(5, &counter);

// 原子条件操作
if (atomic_dec_and_test(&counter))   // 减到 0 返回 true
    pr_info("counter reached zero\n");
```

原子操作在不支持单指令原子性的平台（如早期 ARM）上通过 spinlock 实现，在 x86 上直接映射为 `LOCK` 前缀指令（如 `LOCK XADD`）。

## 信号量 (Semaphore)

信号量是允许固定数量持有者的睡眠锁，在内核中大多已被 mutex 取代，但仍用于计数场景。

```c
#include <linux/semaphore.h>

DEFINE_SEMAPHORE(my_sema);  // 初始值为 1（相当于 mutex）

struct semaphore count_sema;
sema_init(&count_sema, 5); // 初始值 5, 最多 5 个并发访问

down(&count_sema);          // P 操作 (获取)，可睡眠
// ... 访问资源 ...
up(&count_sema);            // V 操作 (释放)

down_interruptible(&count_sema); // 可被信号打断的版本
down_trylock(&count_sema);       // 非阻塞版本
```

## 读写锁 (Read-Write Lock)

区分读和写操作：多个读者可以同时持有锁，写者需要独占。

### rwlock_t (自旋版本)

```c
#include <linux/rwlock.h>

DEFINE_RWLOCK(my_rwlock);

// 读者
read_lock(&my_rwlock);      // 多个读者可同时进入
// ... 读取共享数据 ...
read_unlock(&my_rwlock);

// 写者
write_lock(&my_rwlock);     // 等待所有读者释放，然后独占
// ... 修改共享数据 ...
write_unlock(&my_rwlock);

// 中断安全版本
unsigned long flags;
read_lock_irqsave(&my_rwlock, flags);
write_lock_irqsave(&my_rwlock, flags);
```

### rwsem (睡眠版本)

```c
#include <linux/rwsem.h>

DECLARE_RWSEM(my_rwsem);

// 读者
down_read(&my_rwsem);       // 多个并发读
// ... 读取 ...
up_read(&my_rwsem);

// 写者
down_write(&my_rwsem);      // 独占
// ... 写入 ...
up_write(&my_rwsem);
```

## 内存屏障 (Memory Barrier)

在弱内存模型架构（ARM、PowerPC）中，CPU 和编译器可能重排访存指令。内存屏障强制顺序约束。

```c
#include <asm/barrier.h>

smp_mb();    // 通用内存屏障 (保证所有加载/存储的全局顺序)
smp_rmb();   // 读内存屏障 (保证之前读操作在之后读操作之前完成)
smp_wmb();   // 写内存屏障 (保证之前写操作在之后写操作之前完成)

// 在 x86 上:
//   smp_mb() → mfence 或 LOCK 前缀
//   smp_rmb() → 编译器屏障 (x86 硬件保证读顺序)
//   smp_wmb() → 编译器屏障 (x86 硬件保证写顺序, 除 WC 内存外)
```

更多细节参考 [内存屏障](内存屏障.md)。

## Lockdep (死锁检测)

内核内置的锁依赖验证器。使用时只需在 .config 中启用：

```
CONFIG_LOCKDEP=y
CONFIG_PROVE_LOCKING=y
```

Lockdep 在运行时追踪所有锁的获取顺序，检测：

| 问题 | 含义 |
|------|------|
| AA 死锁 | 同一任务重复获取同一把锁 |
| ABBA 死锁 | 两个任务以不同顺序获取锁 |
| 锁在中断上下文中未关中断 | 在中断中获取了仅在进程上下文持有的锁 |
| 锁依赖反转 | 违反锁顺序约束 |

Lockdep 通过在 `/proc/lockdep_chains` 输出锁依赖图来帮助调试。

## 选择正确的锁: 决策流程

```
临界区很短 (< 几微秒)?
├─ YES → 在中断或软中断上下文中?
│        ├─ YES → spinlock 或 raw_spinlock
│        └─ NO  → spinlock 足够
└─ NO  → 需要保护读多写少的数据?
         ├─ YES → 读者只需要遍历 (不需写)?
         │        ├─ YES → RCU (无锁读)
         │        └─ NO  → rwlock (自旋) 或 rwsem (睡眠)
         └─ NO  → 使用 mutex (进程上下文标准选择)
```
